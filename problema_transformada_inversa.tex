\section*{Función dada}

Para el ejercicio b) se considera:

\begin{equation}
    Y_2(s)=\frac{(s+2)(s+3)}{(s+1)(s-3)(s+5)}.
\end{equation}

Se busca obtener $y_2(t)$ en el dominio del tiempo y representarla gráficamente.

\section*{Descomposición en fracciones simples}

Planteamos:

\begin{equation}
    \frac{(s+2)(s+3)}{(s+1)(s-3)(s+5)}
    =\frac{A}{s+1}+\frac{B}{s-3}+\frac{C}{s+5}.
\end{equation}

Aplicando el método de residuos:

\begin{align}
    A&=\left.\frac{(s+2)(s+3)}{(s-3)(s+5)}\right|_{s=-1}
      =-\frac{1}{8},\\
    B&=\left.\frac{(s+2)(s+3)}{(s+1)(s+5)}\right|_{s=3}
      =\frac{15}{16},\\
    C&=\left.\frac{(s+2)(s+3)}{(s+1)(s-3)}\right|_{s=-5}
      =\frac{3}{16}.
\end{align}

Por lo tanto:

\begin{equation}
    Y_2(s)=-\frac{1}{8}\frac{1}{s+1}
    +\frac{15}{16}\frac{1}{s-3}
    +\frac{3}{16}\frac{1}{s+5}.
\end{equation}

\section*{Expresión en el dominio del tiempo}

Usando

\begin{equation}
    \mathcal{L}^{-1}\left\{\frac{1}{s+a}\right\}=e^{-at}u(t),
\end{equation}

se obtiene:

\begin{equation}
    \boxed{
    y_2(t)=\left(-\frac{1}{8}e^{-t}
    +\frac{15}{16}e^{3t}
    +\frac{3}{16}e^{-5t}\right)u(t)
    }.
\end{equation}

Para $t\geq 0$:

\begin{equation}
    \boxed{
    y_2(t)=-\frac{1}{8}e^{-t}
    +\frac{15}{16}e^{3t}
    +\frac{3}{16}e^{-5t}
    }.
\end{equation}

\section*{Características de la señal}

La señal comienza en:

\begin{equation}
    y_2(0^+)=1.
\end{equation}

Como existe un polo en $s=3$, aparece el término $e^{3t}$ y la respuesta crece
sin límite para $t\to\infty$. Por lo tanto, el sistema es inestable.

\section*{Resultado final}

\begin{equation}
    \boxed{
    Y_2(s)=\frac{(s+2)(s+3)}{(s+1)(s-3)(s+5)}
    \quad\Longleftrightarrow\quad
    y_2(t)=\left(-\frac{1}{8}e^{-t}
    +\frac{15}{16}e^{3t}+\frac{3}{16}e^{-5t}\right)u(t)
    }.
\end{equation}
